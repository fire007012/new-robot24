cmake_minimum_required(VERSION 3.8)
project(host_bridge)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
set(CMAKE_CXX_STANDARD 17)
add_executable(ros_image_rtsp_node src/ros_image_rtsp_node.cpp)
ament_target_dependencies(ros_image_rtsp_node rclcpp sensor_msgs)
install(TARGETS ros_image_rtsp_node DESTINATION lib/${PROJECT_NAME})
file(GLOB bridge_python "${CMAKE_CURRENT_SOURCE_DIR}/*.py")
install(PROGRAMS ${bridge_python} DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS video_stack.sh mediamtx_launcher.sh DESTINATION lib/${PROJECT_NAME})
install(FILES bridge_control.yaml video_sources.example.yaml ros_image_rtsp.example.yaml DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch/ DESTINATION share/${PROJECT_NAME}/launch)
ament_package()
