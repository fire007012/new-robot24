另一个终端 source 同一环境后，显式建立 Mock 会话：

```bash
source /tmp/robot24-upstream-deps-20261108-install/setup.bash
ros2 service call /robot_runtime/init std_srvs/srv/Trigger '{}'
for condition in arm_secured support_trusted payload_retained; do
  ros2 service call "/mock/conditions/$condition" std_srvs/srv/SetBool '{data: true}'
done
ros2 service call /robot_runtime/enable std_srvs/srv/Trigger '{}'
ros2 service call /robot_runtime/resume std_srvs/srv/Trigger '{}'
```
