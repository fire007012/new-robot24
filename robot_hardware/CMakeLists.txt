cmake_minimum_required(VERSION 3.8)
project(robot_hardware)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(canopen_interfaces REQUIRED)
find_package(std_srvs REQUIRED)
find_package(yaml_cpp_vendor REQUIRED)
find_package(diff_drive_controller REQUIRED)
find_package(joint_trajectory_controller REQUIRED)
find_package(joint_state_broadcaster REQUIRED)
foreach(dep controller_interface hardware_interface pluginlib rclcpp rclcpp_lifecycle std_msgs realtime_tools robot_safety)
 find_package(${dep} REQUIRED)
endforeach()
rosidl_generate_interfaces(${PROJECT_NAME} "srv/Maintain.srv" "srv/SetProfile.srv" "srv/GroupOperation.srv")
ament_python_install_package(robot_hardware_runtime)
add_library(robot_controllers SHARED src/timed_velocity_controller.cpp src/session_diff_drive_controller.cpp src/retained_standard_controllers.cpp)
target_include_directories(robot_controllers PUBLIC include)
ament_target_dependencies(robot_controllers controller_interface hardware_interface pluginlib rclcpp rclcpp_lifecycle std_msgs realtime_tools diff_drive_controller joint_trajectory_controller joint_state_broadcaster)
pluginlib_export_plugin_description_file(controller_interface robot_controllers.xml)
add_library(robot_mock_system SHARED src/mock_system.cpp)
target_include_directories(robot_mock_system PRIVATE include)
ament_target_dependencies(robot_mock_system hardware_interface pluginlib rclcpp rclcpp_lifecycle std_srvs canopen_interfaces robot_safety yaml_cpp_vendor)
target_link_libraries(robot_mock_system yaml-cpp ${CMAKE_DL_LIBS})
target_link_libraries(robot_controllers ${CMAKE_DL_LIBS})
pluginlib_export_plugin_description_file(hardware_interface mock_system.xml)
install(TARGETS robot_controllers robot_mock_system LIBRARY DESTINATION lib)
install(PROGRAMS scripts/robot_runtime scripts/robot_group scripts/axis_commission scripts/capture_acceptance DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch urdf config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
 find_package(ament_cmake_gtest REQUIRED)
 ament_add_gtest(test_velocity_watchdog test/test_velocity_watchdog.cpp)
 target_include_directories(test_velocity_watchdog PRIVATE include)
 add_executable(interface_probe test/interface_probe.cpp)
 ament_target_dependencies(interface_probe hardware_interface pluginlib)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_test(NAME test_hardware_topology COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/check_topology.py $<TARGET_FILE:interface_probe>)
 add_test(NAME test_runtime_policy COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/test -p test_*.py -v)
 set_tests_properties(test_runtime_policy PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
ament_export_dependencies(rosidl_default_runtime)
install(DIRECTORY include/ DESTINATION include)
ament_export_include_directories(include)
ament_package()
