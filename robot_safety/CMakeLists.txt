cmake_minimum_required(VERSION 3.8)
project(robot_safety)
find_package(ament_cmake REQUIRED)
add_library(robot_safety SHARED src/domain.cpp)
target_compile_features(robot_safety PUBLIC cxx_std_17)
target_include_directories(robot_safety PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
install(TARGETS robot_safety EXPORT robot_safety_targets LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(robot_safety_targets HAS_LIBRARY_TARGET)
ament_export_libraries(robot_safety)
ament_export_include_directories(include)
if(BUILD_TESTING)
 find_package(ament_cmake_gtest REQUIRED)
 ament_add_gtest(test_domain test/test_domain.cpp)
 target_link_libraries(test_domain robot_safety)
endif()
ament_package()
