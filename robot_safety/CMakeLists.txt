cmake_minimum_required(VERSION 3.8)
project(robot_safety)
find_package(ament_cmake REQUIRED)
find_package(yaml_cpp_vendor REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_srvs REQUIRED)
find_package(std_msgs REQUIRED)
add_library(robot_safety SHARED src/domain.cpp src/policy.cpp src/runtime.cpp)
target_link_libraries(robot_safety yaml-cpp)
ament_target_dependencies(robot_safety rclcpp std_srvs std_msgs)
ament_export_dependencies(yaml_cpp_vendor rclcpp std_srvs std_msgs)
target_compile_features(robot_safety PUBLIC cxx_std_17)
target_include_directories(robot_safety PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
install(TARGETS robot_safety EXPORT robot_safety_targets LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(robot_safety_targets HAS_LIBRARY_TARGET)
ament_export_libraries(robot_safety)
ament_export_include_directories(include)
if(BUILD_TESTING)
 find_package(ament_cmake_gtest REQUIRED)
 ament_add_gtest(test_domain test/test_domain.cpp test/test_group_domain.cpp test/test_freshness.cpp test/test_policy.cpp)
 target_link_libraries(test_domain robot_safety)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_test(NAME test_group_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_group_contract.py)
endif()
install(DIRECTORY config/ DESTINATION share/${PROJECT_NAME}/config)
ament_package()
