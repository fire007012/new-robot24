cmake_minimum_required(VERSION 3.8)
project(robot_camera_sdk)
find_package(ament_cmake REQUIRED)
set(ROBOT24_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.deps/librealsense/opt/ros/humble")
if(NOT EXISTS "${ROBOT24_SDK_ROOT}/include/librealsense2/rs.hpp")
 message(FATAL_ERROR "RealSense SDK missing. Run tools/prepare_camera_sdk.sh first.")
endif()
install(DIRECTORY "${ROBOT24_SDK_ROOT}/include/" DESTINATION include)
install(DIRECTORY "${ROBOT24_SDK_ROOT}/lib/" DESTINATION lib)
ament_environment_hooks(sdk_path.dsv)
ament_package()
