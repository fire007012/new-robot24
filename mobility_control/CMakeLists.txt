cmake_minimum_required(VERSION 3.8)
project(mobility_control)
find_package(ament_cmake REQUIRED)
set(CMAKE_CXX_STANDARD 17)
find_package(ruckig REQUIRED)
foreach(dep rclcpp geometry_msgs std_msgs yaml_cpp_vendor)
  find_package(${dep} REQUIRED)
endforeach()
add_executable(base_cmd_node src/base_cmd_node.cpp)
ament_target_dependencies(base_cmd_node rclcpp geometry_msgs std_msgs yaml_cpp_vendor)
target_link_libraries(base_cmd_node ruckig::ruckig yaml-cpp)
install(TARGETS base_cmd_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch/ DESTINATION share/${PROJECT_NAME}/launch)
install(DIRECTORY config/ DESTINATION share/${PROJECT_NAME}/config)
ament_package()
