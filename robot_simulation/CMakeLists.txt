cmake_minimum_required(VERSION 3.8)
project(robot_simulation)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp hardware_interface pluginlib robot_hardware gz_ros2_control)
 find_package(${dep} REQUIRED)
endforeach()
find_package(ignition-gazebo6 REQUIRED)
add_library(robot24_gazebo_system SHARED src/gazebo_system.cpp)
target_include_directories(robot24_gazebo_system PRIVATE ${robot_hardware_INCLUDE_DIRS})
ament_target_dependencies(robot24_gazebo_system rclcpp hardware_interface pluginlib robot_hardware gz_ros2_control)
target_link_libraries(robot24_gazebo_system ignition-gazebo6::ignition-gazebo6)
pluginlib_export_plugin_description_file(gz_ros2_control plugins.xml)
install(TARGETS robot24_gazebo_system LIBRARY DESTINATION lib)
install(DIRECTORY launch urdf worlds DESTINATION share/${PROJECT_NAME})
ament_package()
